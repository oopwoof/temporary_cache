/* ------------------------------------------------------------ HyperLogLog
   与 data/hll.py 逐位一致的实现。人数和分位数一样不能把每格的标量相加:
   同一个人今天来南京路、明天来环球港, 求和会算成 3 个人。所以每条记录存的是
   可合并的草图, 任意切片下先合并再估算。
   注意 Math.imul —— 32 位乘法用普通 * 会丢精度, 结果与 Python 端对不上。 */
const HLL_B = 10, HLL_M = 1 << HLL_B, HLL_TAIL = 32 - HLL_B;
const HLL_ALPHA = 0.7213 / (1 + 1.079 / HLL_M);

function hllDecode(b64) {
  const bin = atob(b64);
  const regs = new Uint8Array(HLL_M);
  for (let i = 0; i < HLL_M; i++) regs[i] = bin.charCodeAt(i);
  return regs;
}
function hllNew() { return new Uint8Array(HLL_M); }
function hllMerge(target, other) {          // 并集 = 寄存器逐位取最大
  for (let i = 0; i < HLL_M; i++) if (other[i] > target[i]) target[i] = other[i];
}
function hllEstimate(regs) {
  let zeros = 0, inv = 0;
  for (let i = 0; i < HLL_M; i++) {
    const r = regs[i];
    if (r === 0) zeros++;
    inv += Math.pow(2, -r);
  }
  const est = HLL_ALPHA * HLL_M * HLL_M / inv;
  // 小基数时 HLL 的估计量偏差大, 改用线性计数
  if (est <= 2.5 * HLL_M && zeros > 0) return HLL_M * Math.log(HLL_M / zeros);
  return est;
}

/* 从直方图线性插值求分位数 —— 与 generate_mock.py 中的算法一致,
   保证任意日期 × 点位切片下 P95 都能正确聚合(平均值可以加权, 分位数不能)。 */
function percentileFromHist(hist, q) {
  const total = hist.reduce((a, b) => a + b, 0);
  if (!total) return 0;
  const target = q * total; let cum = 0;
  for (let i = 0; i < hist.length; i++) {
    if (cum + hist[i] >= target && hist[i] > 0) {
      const lo = EDGES[i], hi = EDGES[i + 1];
      return Math.round((lo + (hi - lo) * (target - cum) / hist[i]) * 10) / 10;
    }
    cum += hist[i];
  }
  return EDGES[EDGES.length - 1];
}

/* 唯一的聚合入口: 看板上每个数字都由它算出, 所以口径只有一份。 */
function computeMetrics(records) {
  const a = {}; SUM_KEYS.forEach(k => a[k] = 0);
  const hist = new Array(EDGES.length - 1).fill(0);
  const hll = hllNew();
  for (const r of records) {
    for (const k of SUM_KEYS) a[k] += r[k];
    for (let i = 0; i < hist.length; i++) hist[i] += r.gen_seconds_hist[i];
    hllMerge(hll, hllDecode(r.participants_hll));
  }
  const safe = (x, y) => y ? x / y : 0;
  /* participants 是人次; 去重人数要靠合并后的草图估算。
     估算值再用两个已知的硬约束夹一下, 否则会出现"去重人数 > 人次"
     (人均 0.98 次) 这种读者一眼就知道不对的数:
       上界 = 人次总和           —— 一个人至少贡献一次参与
       下界 = 单格去重人数的最大值 —— 并集不可能小于其中任何一个子集
     只有一条记录时不用估, 直接取那天那个点位的精确去重数。 */
  let uniqueVisitors = 0;
  if (records.length === 1) {
    uniqueVisitors = records[0].participants_unique;
  } else if (records.length > 1) {
    let floor = 0;
    for (const r of records) if (r.participants_unique > floor) floor = r.participants_unique;
    uniqueVisitors = Math.min(Math.max(hllEstimate(hll), floor), a.participants);
  }
  return Object.assign(a, {
    hist,
    uniqueVisitors,
    visitsPerUser: uniqueVisitors ? a.participants / uniqueVisitors : 0,
    days: new Set(records.map(r => r.date)).size,
    successRate: safe(a.gen_success, a.gen_requests),
    failRate: safe(a.gen_failed, a.gen_requests),
    avgGen: safe(a.gen_seconds_total, a.gen_success),
    p95: percentileFromHist(hist, 0.95),
    scanRate: safe(a.qr_scans, a.participants),
    shareRate: safe(a.shares, a.participants),
    passRate: safe(a.review_pass, a.review_total),
    rejectRate: safe(a.review_reject, a.review_total),
    pendingRate: safe(a.review_pending, a.review_total),
    onlineRate: safe(a.online_minutes, a.expected_minutes)
  });
}

function deltaNode(cur, prev, opts) {
  const o = opts || {};
  const n = el("div", "t-delta");
  if (prev == null || !isFinite(prev) || (prev === 0 && o.pp !== true)) {
    n.className = "t-delta flat"; n.textContent = "环比 —"; return n;
  }
  let diff, text;
  if (o.pp) { diff = (cur - prev) * 100; text = (diff >= 0 ? "+" : "") + diff.toFixed(1) + "pp"; }
  else if (o.abs) { diff = cur - prev; text = (diff >= 0 ? "+" : "") + diff.toFixed(1) + o.unit; }
  else { diff = (cur - prev) / prev * 100; text = (diff >= 0 ? "+" : "") + diff.toFixed(1) + "%"; }
  const good = o.upIsBad ? diff < 0 : diff > 0;
  const cls = Math.abs(diff) < 0.05 ? "flat" : (good ? "up" : "down");
  n.className = "t-delta " + cls;
  n.appendChild(el("span", null, (diff > 0 ? "▲" : diff < 0 ? "▼" : "■") + " " + text));
  n.appendChild(el("span", null, "较上一周期"));
  return n;
}
/* 迷你趋势线: 12 个点, 收敛的灰色, 末点用强调色的小圆点。 */
function sparkline(values, accent) {
  const NS = "http://www.w3.org/2000/svg";
  const svg = document.createElementNS(NS, "svg");
  svg.setAttribute("class", "spark"); svg.setAttribute("viewBox", "0 0 120 26");
  svg.setAttribute("preserveAspectRatio", "none"); svg.setAttribute("aria-hidden", "true");
  if (values.length < 2) return svg;
  const step = Math.max(1, Math.ceil(values.length / 12));
  const pts = values.filter((_, i) => i % step === 0 || i === values.length - 1);
  const min = Math.min(...pts), max = Math.max(...pts), span = (max - min) || 1;
  const xy = pts.map((v, i) => [
    2 + i * (116 / (pts.length - 1)),
    23 - ((v - min) / span) * 20
  ]);
  const p = document.createElementNS(NS, "polyline");
  p.setAttribute("fill", "none"); p.setAttribute("stroke", tok("--ink-muted"));
  p.setAttribute("stroke-width", "1.5"); p.setAttribute("stroke-linejoin", "round");
  p.setAttribute("stroke-linecap", "round");
  p.setAttribute("points", xy.map(a => a.join(",")).join(" "));
  svg.appendChild(p);
  const c = document.createElementNS(NS, "circle");
  const last = xy[xy.length - 1];
  c.setAttribute("cx", last[0]); c.setAttribute("cy", last[1]); c.setAttribute("r", "2.6");
  c.setAttribute("fill", accent);
  svg.appendChild(c);
  return svg;
}

function lineWithThreshold(id, daily, key, opts) {
  const t = theme();
  // 端点 = (含阈值的范围 + 留白) 向外取整到 snap 的整数倍, 保证刻度是干净的数
  opts.pad = (v, range, dir) => {
    const span = Math.max(range.max - range.min, Math.abs(opts.threshold) * 0.02) || 1;
    const raw = v + dir * span * 0.10, s = opts.snap;
    return (dir < 0 ? Math.floor(raw / s) : Math.ceil(raw / s)) * s;
  };
  chart(id).setOption({
    backgroundColor: "transparent",
    animationDuration: 240,
    grid: baseGrid(t, { top: 26 }),
    tooltip: Object.assign({ trigger: "axis",
      axisPointer: { type: "line", lineStyle: { color: t.axis, width: 1 }, snap: true },
      formatter: p => '<div style="font-weight:600">' + esc(p[0].axisValue) + "</div>"
        + tipRow(p[0].color, opts.name, opts.fmt(p[0].value))
    }, tipBox(t)),
    xAxis: Object.assign({ type: "category", data: daily.map(d => d.date.slice(5)), boundaryGap: false },
      axisCommon(t), { splitLine: { show: false } }),
    yAxis: Object.assign({ type: "value" }, axisCommon(t), {
      axisLine: { show: false },
      axisLabel: { color: t.muted, fontSize: 11, formatter: opts.axisFmt },
      // 阈值线必须落在可视范围内, 否则"超没超"这件事就看不出来了
      min: v => opts.pad(Math.min(v.min, opts.threshold), v, -1),
      max: v => opts.pad(Math.max(v.max, opts.threshold), v, 1)
    }),
    series: [{
      name: opts.name, type: "line", data: daily.map(d => d[key]),
      symbol: "circle", symbolSize: 8, showSymbol: false,
      lineStyle: { width: 2, color: t.s1, cap: "round", join: "round" },
      itemStyle: { color: t.s1, borderColor: t.surface, borderWidth: 2 },
      areaStyle: { color: t.s1, opacity: 0.10 },
      markLine: {
        silent: true, symbol: "none",
        // 极值标记通常落在右半边, 阈值标签放左端避免压字
        label: { formatter: opts.thresholdLabel, color: t.ink2, fontSize: 11,
          position: "insideStartTop" },
        lineStyle: { color: t.critical, width: 1, type: "dashed" },
        data: [{ yAxis: opts.threshold }]
      },
      markPoint: opts.extreme ? {
        symbol: "circle", symbolSize: 9,
        itemStyle: { color: t.critical, borderColor: t.surface, borderWidth: 2 },
        label: { color: t.ink2, fontSize: 11, position: opts.extremePos || "top",
          formatter: p => opts.fmt(p.value) },
        data: [{ type: opts.extreme }]
      } : undefined
    }]
  }, true);
}

function renderFunnel(m) {
  const t = theme();
  // 五段。原来把"参与"直接连到"生成成功", 会把两类完全不同的流失混在一起:
  // 一类是压根没点生成就走了(引导/兴趣问题), 一类是点了没成功(技术问题)。
  const stages = [
    { name: "参与（触屏开始）", value: m.participants },
    { name: "发起生成", value: m.gen_users },
    { name: "生成成功", value: m.gen_success_users },
    { name: "扫码带走", value: m.qr_scans },
    { name: "分享出去", value: m.shares }
  ];
  const top = stages[0].value || 1;
  chart("funnel").setOption({
    backgroundColor: "transparent",
    animationDuration: 240,
    tooltip: Object.assign({ trigger: "item",
      formatter: p => '<div style="font-weight:600">' + esc(p.name) + "</div>"
        + tipRow(t.ord[p.dataIndex], "人数", fmtInt(p.value) + " 人")
        + tipRow(t.ord[p.dataIndex], "占参与", fmtPct(p.value / top))
    }, tipBox(t)),
    series: [{
      type: "funnel", top: 16, bottom: 16, left: "8%", right: "8%",
      minSize: "36%", maxSize: "94%", sort: "descending",
      gap: 2,                                   // 2px 面色缝隙
      data: stages.map((s, i) => ({ name: s.name, value: s.value,
        itemStyle: { color: t.ord[i], borderWidth: 0 } })),
      label: { show: true, position: "inside", color: "#fff", fontSize: 12, fontWeight: 600,
        formatter: p => p.name + "  " + fmtCompact(p.value) + "（" + fmtPct(p.value / top, 0) + "）" },
      labelLine: { show: false },
      emphasis: { label: { fontSize: 12 } }
    }]
  }, true);
  renderTable("funnel", ["环节", "人数", "占参与人数", "较上一环节"],
    stages.map((s, i) => [s.name, fmtInt(s.value), fmtPct(s.value / top),
      i === 0 ? "—" : fmtPct(s.value / (stages[i - 1].value || 1))]),
    ["", "num", "num", "num"]);
}

function renderHeat(records) {
  const t = theme();
  const openH = META.open_hour, closeH = META.close_hour;
  const hours = []; for (let h = openH; h < closeH; h++) hours.push(h + ":00");
  const sum = {}, dayCount = {};
  const seen = new Set();
  for (const r of records) {
    const wd = weekdayOf(r.date);
    if (!seen.has(r.date + "|" + wd)) { seen.add(r.date + "|" + wd); dayCount[wd] = (dayCount[wd] || 0) + 1; }
    for (let h = openH; h < closeH; h++) {
      const k = wd + "|" + h; sum[k] = (sum[k] || 0) + r.hourly_participants[h];
    }
  }
  const data = [], vals = [];
  for (let wd = 0; wd < 7; wd++) for (let h = openH; h < closeH; h++) {
    const n = dayCount[wd] ? sum[wd + "|" + h] / dayCount[wd] : 0;
    data.push([h - openH, wd, Math.round(n)]); vals.push(n);
  }
  const max = Math.max(1, ...vals);
  chart("heat").setOption({
    backgroundColor: "transparent",
    animationDuration: 240,
    grid: { left: 8, right: 16, top: 12, bottom: 56, containLabel: true },
    tooltip: Object.assign({ trigger: "item",
      formatter: p => '<div style="font-weight:600">' + WEEKDAYS[p.value[1]] + " " + hours[p.value[0]] + "</div>"
        + tipRow(t.s1, "日均参与", fmtInt(p.value[2]) + " 人")
    }, tipBox(t)),
    xAxis: { type: "category", data: hours, splitArea: { show: false },
      axisLine: { lineStyle: { color: t.axis } }, axisTick: { show: false },
      axisLabel: { color: t.muted, fontSize: 11 } },
    yAxis: { type: "category", data: WEEKDAYS, inverse: true, splitArea: { show: false },
      axisLine: { lineStyle: { color: t.axis } }, axisTick: { show: false },
      axisLabel: { color: t.ink2, fontSize: 12 } },
    visualMap: {
      min: 0, max: Math.ceil(max), calculable: true, orient: "horizontal",
      left: "center", bottom: 8, itemWidth: 12, itemHeight: 120,
      // 连续量 = 单一色相, 浅→深; 绝不用彩虹色
      inRange: { color: [t.seqLo, t.seqHi] },
      textStyle: { color: t.muted, fontSize: 11 },
      text: ["人多", "人少"]
    },
    series: [{
      type: "heatmap", data: data,
      itemStyle: { borderColor: t.surface, borderWidth: 2 },  // 2px 面色缝隙
      emphasis: { itemStyle: { borderColor: t.ink2, borderWidth: 2 } },
      progressive: 0
    }]
  }, true);
  renderTable("heat", ["星期"].concat(hours),
    WEEKDAYS.map((w, wd) => [w].concat(hours.map((_, hi) =>
      fmtInt(data.find(d => d[0] === hi && d[1] === wd)[2])))),
    [""].concat(hours.map(() => "num")));
}

    text: m => "当前 " + fmtPct(m.scanRate) + "，低于 " + fmtPct(TH.scan_rate_min, 0) + " 阈值" }
];
const SEV_ORDER = { critical: 0, serious: 1, warning: 2 };

function renderAlerts(perDevice, records) {
  const box = $("#alerts"); box.textContent = "";
  const byDevDate = {};
  for (const r of records) (byDevDate[r.device_id] = byDevDate[r.device_id] || []).push(r);
  // 整个数据集最后一天的指标 —— 用来判断"到现在为止恢复了没有",
  // 这和"所选范围的最后一天"是两件事。
  const latestByDevice = {};
  for (const r of DAILY) if (r.date === MAX_DATE) latestByDevice[r.device_id] = computeMetrics([r]);
  const out = [];
  for (const d of perDevice) {
    const recs = byDevDate[d.id] || [];
    const perDay = recs.map(r => ({ date: r.date, m: computeMetrics([r]) }));
    const last = perDay[perDay.length - 1];
    for (const rule of RULES) {
      const aggHit = rule.hit(d.m);
      const lastHit = last ? rule.hit(last.m) : false;
      if (!aggHit && !lastHit) continue;
      const days = perDay.filter(x => rule.hit(x.m)).length;
      // 两件不同的事, 之前用同一个词表达是有歧义的:
      //   endHit —— 所选范围的最后一天越界(筛历史区间时, 这不代表现在还没好)
      //   unresolved —— 拿整个数据集的最后一天比, 才是"截至最新数据仍未恢复"
      const latest = latestByDevice[d.id];
      const unresolved = latest ? rule.hit(latest) : false;
      out.push({
        dev: d, rule: rule, days: days,
        endHit: lastHit, unresolved: unresolved,
        urgent: unresolved,
        scope: aggHit ? "区间合计" : "范围末日（" + last.date + "）",
        text: aggHit ? rule.text(d.m) : rule.text(last.m)
      });
    }
  }
  // 还在发生的排前面, 再按严重程度, 再按越界天数
  out.sort((a, b) => (b.unresolved - a.unresolved) || (b.endHit - a.endHit)
    || (SEV_ORDER[a.rule.sev] - SEV_ORDER[b.rule.sev]) || (b.days - a.days));
  if (!out.length) {
    box.appendChild(el("p", "empty", "当前筛选范围内所有指标都在阈值内，没有需要处理的预警。"));
    return;
  }
  const unres = out.filter(a => a.unresolved).length;
  const head = el("p", "legend-note", "共 " + out.length + " 条，其中 " + unres
    + " 条截至最新数据（" + MAX_DATE + "）仍未恢复，排在最前；其余为所选范围内的历史越界。"
    + "级别以文字标注，不单靠颜色区分。");
  box.appendChild(head);
  const list = el("div", "alert-list");
  out.forEach(a => {
    const n = el("div", "alert-item " + a.rule.sev);
    const h = el("div", "a-head");
    const s = el("span", "sev sev-" + a.rule.sev);
    s.appendChild(el("span", "dot"));
    s.appendChild(el("span", null, SEV_LABEL[a.rule.sev]));
    h.appendChild(s);
    h.appendChild(el("span", null, a.dev.name));
    h.appendChild(el("span", "pill", a.rule.label));
    if (a.unresolved) h.appendChild(el("span", "pill open", "截至 " + MAX_DATE + " 未恢复"));
    else if (a.endHit) h.appendChild(el("span", "pill", "范围末日越界"));
    n.appendChild(h);
    n.appendChild(el("div", "a-body",
      a.scope + "：" + a.text + "；范围内有 " + a.days + " 天单日越界。"));
    n.appendChild(el("div", "a-act", "建议：" + (a.rule.action || ACTION[a.rule.type])));
    list.appendChild(n);
  });
  box.appendChild(list);
}

function renderDevices(perDevice, lastDay) {
  const wrap = $("#device-table"); wrap.textContent = "";
  const isLatest = lastDay === MAX_DATE;

  /* 状态的时间基准必须和同一行的"当日指标"一致。
     - 选到数据集最新一天: 用设备上报的实时状态(心跳), 它能反映"此刻离线"——
       某台设备 20:32 才断网, 当天在线率仍有 87.5%, 按日聚合是推不出来的。
     - 选到历史日期: 实时心跳与那天无关, 改由那天的数据推导, 心跳列置空。
     之前的写法是无论筛到哪天都显示实时快照, 和当日指标不是同一个时间点。 */
  function statusOn(rec) {
    if (!rec) return "no-data";
    const m = computeMetrics([rec]);
    if (m.onlineRate < 0.5) return "offline";
    if ((m.gen_requests && m.successRate < TH.success_rate_min) || m.avgGen > TH.avg_gen_seconds_max)
      return "degraded";
    return "online";
  }

  const rows = perDevice.map(r => {
    const rec = DAILY.find(x => x.date === lastDay && x.device_id === r.id);
    return { r: r, rec: rec, m: rec ? computeMetrics([rec]) : null,
             st: isLatest ? r.dev.status : statusOn(rec) };
  });
  const order = { offline: 0, degraded: 1, online: 2, "no-data": 3 };
  rows.sort((a, b) => order[a.st] - order[b.st] || a.r.id.localeCompare(b.r.id));

  const bad = rows.filter(x => x.st === "offline" || x.st === "degraded").length;
  $("#device-sub").textContent = rows.length + " 个点位在筛选范围内；下表为 " + lastDay
    + (isLatest ? "（数据集最新一天，状态取自设备心跳）" : "（历史日期，状态由当天数据推导，非实时）")
    + " 的状态与当日指标，"
    + (bad ? bad + " 个当天不可正常服务" : "当天全部正常");

  const table = el("table", "tbl");
  const head = ["状态（" + lastDay + "）", "点位", "城市", "机型",
    isLatest ? "最后心跳" : "最后心跳（仅最新日可用）",
    "当日参与人次", "当日成功率", "当日平均时长", "范围内在线率"];
  const aligns = ["", "", "", "", "", "num", "num", "num", "num"];
  const thead = el("thead"), tr = el("tr");
  head.forEach((h, i) => tr.appendChild(el("th", aligns[i] === "num" ? "num" : null, h)));
  thead.appendChild(tr); table.appendChild(thead);
  const tb = el("tbody");
  rows.forEach(x => {
    const t1 = x.m, tr2 = el("tr");
    const st = el("td");
    const sp = el("span", "status s-" + (x.st === "no-data" ? "offline" : x.st));
    sp.appendChild(el("span", "dot"));
    sp.appendChild(el("span", null, STATUS_LABEL[x.st] || "无数据"));  // 图标 + 文字, 不靠颜色单独表意
    st.appendChild(sp); tr2.appendChild(st);
    [x.r.name, x.r.city, x.r.dev.model].forEach(v => tr2.appendChild(el("td", null, v)));
    tr2.appendChild(el("td", null, isLatest ? x.r.dev.last_heartbeat : "—"));
    tr2.appendChild(el("td", "num", t1 ? fmtInt(t1.participants) : "—"));
    tr2.appendChild(el("td", "num", t1 && t1.gen_requests ? fmtPct(t1.successRate) : "—"));
    tr2.appendChild(el("td", "num", t1 && t1.gen_success ? fmtSec(t1.avgGen) : "—"));
    tr2.appendChild(el("td", "num", fmtPct(x.r.m.onlineRate)));
    tb.appendChild(tr2);
  });
  table.appendChild(tb); wrap.appendChild(table);
}

    set.has(e.device_id)
    && e.start_date <= state.end && e.end_date >= state.start
    && (!f.sev || e.severity === f.sev)
    && (!f.type || e.type === f.type)
    && (!f.status || e.status === f.status));

  }

    r.addEventListener("keydown", ev => { if (ev.key === "Enter" || ev.key === " ") { ev.preventDefault(); toggle(); } });
    tb.appendChild(r); tb.appendChild(detail);
  });
  table.appendChild(tb); wrap.appendChild(table);
}

/* ------------------------------------------------------------ 主渲染 */
function render() {
  const ids = activeDeviceIds();
  const records = slice(state.start, state.end, ids);
  const span = daysBetween(state.start, state.end) + 1;
  const prevEnd = addDays(state.start, -1), prevStart = addDays(prevEnd, -(span - 1));
  const prevRecords = prevStart >= MIN_DATE ? slice(prevStart, prevEnd, ids) : [];

  const cur = computeMetrics(records);
  const prev = prevRecords.length ? computeMetrics(prevRecords) : {};

  // 按日聚合(跨选中的点位)
  const byDate = {};
  for (const r of records) (byDate[r.date] = byDate[r.date] || []).push(r);
  const daily = Object.keys(byDate).sort().map(d => Object.assign({ date: d }, computeMetrics(byDate[d])));

  // 按点位聚合
  const byDev = {};
  for (const r of records) (byDev[r.device_id] = byDev[r.device_id] || []).push(r);
  const perDevice = ids.filter(id => byDev[id]).map(id => ({
    id: id, name: DEV_BY_ID[id].name, city: DEV_BY_ID[id].city,
    dev: DEV_BY_ID[id], m: computeMetrics(byDev[id])
  }));

  $("#scope-note").textContent = "当前范围：" + state.start + " 至 " + state.end
    + "（共 " + span + " 天）· " + perDevice.length + " 个点位 · "
    + (prevRecords.length ? "环比对照 " + prevStart + " 至 " + prevEnd : "环比对照数据不足");

  renderKPIs(cur, prev, daily);
  renderAlerts(perDevice, records);
  renderTrend(daily);
  renderSuccess(daily);
  renderLatency(daily);
  renderReview(daily);
  renderRank(perDevice);
  renderFunnel(cur);
  renderHeat(records);
  renderDevices(perDevice, state.end);
  renderIncidents(ids);
}

/* ------------------------------------------------------------ 控件 */
function setPreset(id) {
  const p = PRESETS.find(x => x.id === id);
  state.preset = id;
  state.end = MAX_DATE;
  state.start = clampDate(addDays(MAX_DATE, -(p.days - 1)));
  $("#date-start").value = state.start;
  $("#date-end").value = state.end;
  syncPresetButtons();
}
function syncPresetButtons() {
  document.querySelectorAll("#presets .btn").forEach(b =>
    b.setAttribute("aria-pressed", String(b.dataset.preset === state.preset)));
}
function buildControls() {
  const pr = $("#presets");
  PRESETS.forEach(p => {
    const b = el("button", "btn", p.label);
    b.type = "button"; b.dataset.preset = p.id;
    b.addEventListener("click", () => { setPreset(p.id); render(); });
    pr.appendChild(b);
  });
  const ds = $("#date-start"), de = $("#date-end");
  [ds, de].forEach(i => { i.min = MIN_DATE; i.max = MAX_DATE; });
  ds.value = state.start; de.value = state.end;
  ds.addEventListener("change", () => {
    state.start = clampDate(ds.value || state.start);
    if (state.start > state.end) state.end = state.start;
    de.value = state.end; state.preset = ""; syncPresetButtons(); render();
  });
  de.addEventListener("change", () => {
    state.end = clampDate(de.value || state.end);
    if (state.end < state.start) state.start = state.end;
    ds.value = state.start; state.preset = ""; syncPresetButtons(); render();
  });

  const city = $("#city");

## `data/hll.py`

HyperLogLog 实现。参与人数跨天跨点位去重的基础 —— 人数与分位数一样不可加，所以每格存可合并的草图而非标量。与 index.html 里的 JS 版必须逐位一致。

共 99 行，3,278 字节。

```python
# -*- coding: utf-8 -*-
"""
HyperLogLog 基数估算 —— 让"参与人数"可以跨天、跨点位去重。

为什么需要它
------------
daily 是按 天 × 点位 聚合的。人数和分位数一样, **不能**把每格算好的标量相加:
同一个人今天来南京路、明天来环球港, 直接求和会把他算成 3 个人。
所以每格存的不是一个数, 而是一个可合并的草图 —— 任意日期 × 点位切片下,
先合并草图再估算, 才是该切片真正的去重人数。
(这与 gen_seconds_hist 存直方图而不是存 P95 是同一个思路: 可加的存标量,
 不可加的存结构。)

规格 —— 必须与 index.html 中的 JS 实现逐位一致
------------------------------------------------
  b = 10 → m = 1024 个寄存器, 理论标准误 1.04/√m ≈ 3.25%
  哈希 = 32 位 FNV-1a + murmur3 finalizer(fmix32)
  高 b 位取寄存器下标, 低 32-b 位数前导零 +1, 寄存器取最大值
  小基数走线性计数, 大基数走标准 HLL + alpha 修正
  寄存器按 1 字节存, base64 编码后进 JSON

关于 finalizer: 不能省。裸 FNV-1a 对 "u0000001" 这类连续键混合不足,
实测 n=300 时估算偏低 49%, 寄存器下标卡方 2473(理想 1023)。
加 fmix32 后最大误差 3.9%, 卡方 1093。

寄存器为什么按 1 字节而不是 4 bit 打包: 22 位尾部的理论最大 rank 是 23,
全量十几万次抽样中预计有几次超过 15, 打包会静默截断。
"""

import base64
import math

HLL_B = 10
HLL_M = 1 << HLL_B           # 1024
_TAIL_BITS = 32 - HLL_B      # 22
_ALPHA = 0.7213 / (1 + 1.079 / HLL_M)
_LINEAR_COUNTING_CUTOFF = 2.5 * HLL_M


def hash32(s: str) -> int:
    """32 位 FNV-1a, 再过一遍 murmur3 的 fmix32 做雪崩。"""
    h = 0x811C9DC5
    for ch in s.encode("utf-8"):
        h ^= ch
        h = (h * 0x01000193) & 0xFFFFFFFF
    h ^= h >> 16
    h = (h * 0x85EBCA6B) & 0xFFFFFFFF
    h ^= h >> 13
    h = (h * 0xC2B2AE35) & 0xFFFFFFFF
    h ^= h >> 16
    return h


def hll_new():
    return [0] * HLL_M


def hll_add(regs, item: str) -> None:
    h = hash32(item)
    idx = h >> _TAIL_BITS                       # 高 10 位 = 寄存器下标
    tail = (h << HLL_B) & 0xFFFFFFFF            # 低 22 位左移到高位, 便于数前导零
    if tail == 0:
        rank = _TAIL_BITS + 1
    else:
        rank = 1
        while not tail & 0x80000000:
            tail = (tail << 1) & 0xFFFFFFFF
            rank += 1
    if rank > regs[idx]:
        regs[idx] = rank


def hll_merge(target, other) -> None:
    """并集 = 寄存器逐位取最大。这就是"可合并"的全部含义。"""
    for i in range(HLL_M):
        if other[i] > target[i]:
            target[i] = other[i]


def hll_estimate(regs) -> float:
    zeros = 0
    inv = 0.0
    for r in regs:
        if r == 0:
            zeros += 1
        inv += 2.0 ** -r
    est = _ALPHA * HLL_M * HLL_M / inv
    if est <= _LINEAR_COUNTING_CUTOFF and zeros > 0:
        # 小基数时 HLL 的估计量偏差大, 改用线性计数(按空寄存器占比反推)
        return HLL_M * math.log(HLL_M / zeros)
    return est


def hll_encode(regs) -> str:
    return base64.b64encode(bytes(regs)).decode("ascii")


def hll_decode(b64: str):
    return list(base64.b64decode(b64))
```

def draw_users(dev_id: str, city: str, n: int):
    """抽出这天这个点位的 n 次参与分别是谁。

    绝大多数是第一次来的新人, 少数是本点位回头客, 极少数是同城其他点位来过的人。
    返回的是**人次**列表, 里面可以有重复 —— 同一个人一天内玩两次是正常的。
    """
    seen_dev = _seen_by_device[dev_id]
    seen_city = _seen_by_city.setdefault(city, [])
    out = []
    for _ in range(n):
        r = rng.random()
        if r < REPEAT_SAME_DEVICE and seen_dev:
            lo = max(0, len(seen_dev) - RECENT_POOL)
            out.append(seen_dev[rng.randrange(lo, len(seen_dev))])
        elif r < REPEAT_SAME_DEVICE + REPEAT_SAME_CITY and seen_city:
            lo = max(0, len(seen_city) - RECENT_POOL)
            uid = seen_city[rng.randrange(lo, len(seen_city))]
            out.append(uid)
            seen_dev.append(uid)
        else:
            uid = _new_uid()
            out.append(uid)
            seen_dev.append(uid)
            seen_city.append(uid)
    return out


def hist_bucket(seconds: float) -> int:
    for i in range(len(HIST_EDGES) - 1):
        if seconds < HIST_EDGES[i + 1]:
            return i
    return len(HIST_EDGES) - 2


def percentile_from_hist(hist, q=0.95):
    """从直方图线性插值估算分位数 —— 保证任意日期/点位切片下 P95 都可聚合还原。"""
    total = sum(hist)
    if total == 0:
        return 0.0
    target = q * total
    cum = 0
    for i, c in enumerate(hist):
        if cum + c >= target and c > 0:
            lo, hi = HIST_EDGES[i], HIST_EDGES[i + 1]
            return round(lo + (hi - lo) * (target - cum) / c, 1)
        cum += c
    return float(HIST_EDGES[-1])


def gen_duration(slow_factor: float, failed: bool) -> float:
    """成功生成的耗时 ~ 对数正态; slow_factor 模拟排队/降级时的整体拉长。"""
    base = rng.lognormvariate(math.log(6.2), 0.42) * slow_factor
    if failed:                       # 失败样本不计入平均时长, 这里只用于超时判定
        base *= rng.uniform(1.8, 4.0)
    return round(min(base, 119.0), 2)

        for dev_id, _name, city, base, model in DEVICES:

            gen_users = int(participants * rng.uniform(0.70, 0.82))

            # ---------- 生成时长 ----------
            slow = 1.0
            if dev_id == SLOW_DEVICE and is_weekend:
                slow = rng.uniform(1.8, 2.3)                    # 周末排队

                "review_reject": review_reject,
                "review_pending": review_pending,
                "online_minutes": online_minutes,
                "expected_minutes": EXPECTED_MINUTES,
                "hourly_participants": hourly,
            })

def _emit_breach(add, info, dev, etype, ongoing=False):
    """把一段连续越界写成一条事件: 未恢复的标 open, 已恢复的标 resolved。"""
    days = info["days"]
    span = f"持续 {days} 天" if days > 1 else "当日"
    detail = f"{info['text']} ({span}, {info['start'].isoformat()} 至 {info['last'].isoformat()})"
    add(info["start"], dev, etype, info["sev"], detail,
        "open" if ongoing else "resolved", hour=info["start"].weekday() % 6 + 11,
        end_day=info["last"])


def build_incidents(records):
    """异常事件: 一部分由上面的故事线必然产生, 一部分是日常随机噪声。"""
    events = []
    by_key = {(r["date"], r["device_id"]): r for r in records}
    start = LAST_DAY - timedelta(days=DAYS - 1)

    def add(day, dev, etype, severity, detail, status, hour=None, minute=None, end_day=None):
        """一条异常事件。

        持续型异常必须带 end_date —— 只记开始日的话, 用户把日期筛到事故中段
        就查不到它了(预警在报、事件表却空着)。看板按"区间相交"过滤。
        """
        t = datetime.combine(day, time(
            hour if hour is not None else rng.randint(OPEN_HOUR, CLOSE_HOUR - 1),
            minute if minute is not None else rng.randint(0, 59),
        ))
        ongoing = status == "open"
        events.append({
            "id": f"INC-{len(events) + 1:04d}",
            "time": t.strftime("%Y-%m-%d %H:%M"),
            "date": day.isoformat(),
            "start_date": day.isoformat(),
            # 未恢复的事件一直延续到数据最后一天
            "end_date": (LAST_DAY if ongoing else (end_day or day)).isoformat(),
            "ongoing": ongoing,
            "device_id": dev,
            "type": etype,
            "severity": severity,
            "detail": detail,
            "status": status,
        })

    # 故事 1: 整机离线
    off_start, off_end = min(OFFLINE_DAYS), max(OFFLINE_DAYS)
    add(off_start, OFFLINE_DEVICE, "device_offline", "critical",
        f"机柜断电导致整机离线, 全天 0 参与 (持续 {len(OFFLINE_DAYS)} 天, "
        f"{off_start.isoformat()} 至 {off_end.isoformat()})",
        "resolved", hour=9, minute=41, end_day=off_end)
    add(off_start, OFFLINE_DEVICE, "heartbeat_lost", "serious",
        "连续 3 次心跳未上报, 判定为离线", "resolved", hour=9, minute=44, end_day=off_end)

    # 故事 6: 最后一天晚间断网, 到现在还没恢复
    add(LAST_DAY, LIVE_OFFLINE_DEVICE, "device_offline", "critical",
        "20:32 起心跳中断, 大屏无响应, 当晚剩余营业时段全部损失", "open", hour=20, minute=32)
    add(LAST_DAY, LIVE_OFFLINE_DEVICE, "network_jitter", "serious",
        "断网前 10 分钟上行丢包率持续升高", "open", hour=20, minute=21)

    # 故事 2 / 3 / 4: 指标越界按"连续区间"开单(和真实告警系统一样做抑制),
    # 同一台设备的同一类异常连续多天只产生 1 条事件, 详情里写明持续天数。
    # 逐日的越界情况由看板前端的规则引擎实时扫描, 不在这里重复堆事件。
    breaches = {"model_error": {}, "gen_timeout": {}, "review_reject": {}}
    for r in sorted(records, key=lambda x: (x["device_id"], x["date"])):
        day = date.fromisoformat(r["date"])
        if day < start or r["participants"] == 0:
            continue
        success_rate = r["gen_success"] / r["gen_requests"] if r["gen_requests"] else 1.0
        avg = r["gen_seconds_total"] / r["gen_success"] if r["gen_success"] else 0.0
        reject_rate = r["review_reject"] / r["review_total"] if r["review_total"] else 0.0
        hit = {
            "model_error": (r["gen_requests"] > 0 and success_rate < 0.90,
                            "critical", f"生成成功率 {success_rate:.1%}, 低于 90% 阈值"),
            "gen_timeout": (avg > 12, "serious", f"平均生成时长 {avg:.1f}s, 超过 12s 阈值"),
            "review_reject": (r["review_total"] > 0 and reject_rate > 0.15,
                              "warning", f"审核拒绝率 {reject_rate:.1%}, 超过 15% 阈值"),
        }
        for etype, (over, sev, text) in hit.items():
            streak = breaches[etype]
            prev = streak.get(r["device_id"])
            if over:
                if prev and prev["last"] == day - timedelta(days=1):
                    prev["last"] = day          # 仍在同一次异常里, 只延长不新开单
                    prev["days"] += 1
                    prev["text"] = text
                else:
                    streak[r["device_id"]] = {"start": day, "last": day, "days": 1,
                                              "sev": sev, "text": text, "emitted": None}
            elif prev:
                streak.pop(r["device_id"])
                _emit_breach(add, prev, r["device_id"], etype)
    for etype, streak in breaches.items():
        for dev, info in streak.items():
            _emit_breach(add, info, dev, etype, ongoing=True)

    # 日常随机噪声
    for _ in range(26):
        day = start + timedelta(days=rng.randint(0, DAYS - 1))
        dev = rng.choice(DEVICES)[0]
        etype, _label, detail = rng.choice([
            e for e in INCIDENT_LIBRARY
            if e[0] in ("asset_download", "printer_error", "network_jitter", "heartbeat_lost")
        ])
        sev = rng.choice(["warning", "warning", "serious"])
        rec = by_key.get((day.isoformat(), dev))
        if rec and rec["participants"] == 0:
            continue
        add(day, dev, etype, sev, detail,
            "open" if day >= LAST_DAY - timedelta(days=2) and rng.random() < 0.6 else "resolved")

    events.sort(key=lambda e: e["time"], reverse=True)
    for i, e in enumerate(events, 1):
        e["id"] = f"INC-{i:04d}"
    return events


def build_devices(records):
    """设备当前状态: 由最后一天的实际数据推导, 不另外编造。"""
    last = {r["device_id"]: r for r in records if r["date"] == LAST_DAY.isoformat()}

    for dev_id, name, city, _base, model in DEVICES:

        avg = r["gen_seconds_total"] / r["gen_success"] if r["gen_success"] else 0.0
        online_rate = r["online_minutes"] / r["expected_minutes"]
        if dev_id == LIVE_OFFLINE_DEVICE:
            status, gap = "offline", 86          # 21:58 - 86min ≈ 20:32 断的心跳
        elif online_rate < 0.5:
            status, gap = "offline", rng.randint(180, 600)
        elif success_rate < 0.90 or avg > 12:
            status, gap = "degraded", rng.randint(1, 4)
        else:
            status, gap = "online", rng.randint(0, 3)
        heartbeat = datetime.combine(LAST_DAY, time(21, 58)) - timedelta(minutes=gap)

def measure_hll_accuracy(records, truth, trials=200):
    """随机切片上实测 HLL 估算误差。

    文档里引用的是这里测出来的数, 不是 1.04/sqrt(m) 这个理论值 ——
    理论标准误假设哈希理想均匀, 实际误差要靠对账才知道。
    """
    by_key = {(r["date"], r["device_id"]): r for r in records}
    dates = sorted({r["date"] for r in records})
    dev_ids = [d[0] for d in DEVICES]
    check = random.Random(20260916)
    errs = []
    for _ in range(trials):
        i = check.randrange(len(dates))
        j = check.randrange(i, len(dates))
        devs = check.sample(dev_ids, check.randint(1, len(dev_ids)))
        keys = [(d, dv) for d in dates[i:j + 1] for dv in devs]
        real = set()
        regs = hll_new()
        for k in keys:
            r = by_key.get(k)
            if not r:
                continue
            real |= truth[k]
            hll_merge(regs, hll_decode(r["participants_hll"]))
        if len(real) < 50:            # 基数太小时相对误差没有参考意义
            continue
        errs.append(abs(hll_estimate(regs) - len(real)) / len(real))
    errs.sort()
    return {
        "slices_tested": len(errs),
        "median_abs_err_pct": round(statistics.median(errs) * 100, 2),
        "p95_abs_err_pct": round(errs[int(len(errs) * 0.95)] * 100, 2),
        "max_abs_err_pct": round(errs[-1] * 100, 2),
        "registers": HLL_M,
        "theoretical_stderr_pct": round(104 / (HLL_M ** 0.5), 2),
    }

# -*- coding: utf-8 -*-
"""
指标交叉校验 —— 完全绕开前端, 独立重算一遍看板上的每个数字。

存在的意义: index.html 里的 computeMetrics() 是唯一的聚合入口, 但"唯一"不等于"正确"。
这个脚本用另一套实现算同样的口径, 两边对不上就说明至少有一边错了。

用法:
    python3 scripts/verify_metrics.py            # 打印各场景期望值(JSON)
    python3 scripts/verify_metrics.py --report   # 人看的报告, 含 HLL 精度实测
"""

import json
import math
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent

from hll import hll_decode, hll_estimate, hll_merge, hll_new   # noqa: E402

D = json.loads((ROOT / "data" / "mock.json").read_text(encoding="utf-8"))
META, EDGES = D["meta"], D["meta"]["hist_edges"]
TH = META["thresholds"]

SUM_KEYS = ["participants", "participants_unique", "gen_users", "gen_requests",
            "gen_success", "gen_failed", "gen_seconds_total", "gen_success_users",
            "qr_scans", "shares", "review_total", "review_pass", "review_reject",
            "review_pending", "online_minutes", "expected_minutes"]


def percentile_from_hist(hist, q=0.95):
    """与前端 percentileFromHist() 同算法。注意这是**估算**, 不是精确分位数。"""
    total = sum(hist)
    if not total:
        return 0.0
    target, cum = q * total, 0
    for i, c in enumerate(hist):
        if cum + c >= target and c > 0:
            lo, hi = EDGES[i], EDGES[i + 1]
            return round(lo + (hi - lo) * (target - cum) / c, 1)
        cum += c
    return float(EDGES[-1])


def compute(records):
    a = {k: sum(r[k] for r in records) for k in SUM_KEYS}
    hist = [0] * (len(EDGES) - 1)
    regs = hll_new()
    for r in records:
        for i, c in enumerate(r["gen_seconds_hist"]):
            hist[i] += c
        hll_merge(regs, hll_decode(r["participants_hll"]))
    safe = lambda x, y: (x / y) if y else 0.0
    # 与前端同样的夹逼: 上界是人次, 下界是单格去重的最大值; 单条记录直接取精确值
    if len(records) == 1:
        uniq = float(records[0]["participants_unique"])
    elif records:
        floor = max(r["participants_unique"] for r in records)
        uniq = min(max(hll_estimate(regs), floor), float(a["participants"]))
    else:
        uniq = 0.0
    return dict(a,
                days=len({r["date"] for r in records}),
                uniqueVisitors=uniq,
                visitsPerUser=safe(a["participants"], uniq),
                successRate=safe(a["gen_success"], a["gen_requests"]),
                avgGen=safe(a["gen_seconds_total"], a["gen_success"]),
                p95=percentile_from_hist(hist),
                scanRate=safe(a["qr_scans"], a["participants"]),
                shareRate=safe(a["shares"], a["participants"]),
                passRate=safe(a["review_pass"], a["review_total"]),
                rejectRate=safe(a["review_reject"], a["review_total"]),
                onlineRate=safe(a["online_minutes"], a["expected_minutes"]))


def sel(start, end, devices=None):

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
docs/*.template.md  →  docs/*.md

存在的意义: 阈值这类数字如果在数据里写一遍、在文档里再手写一遍, 迟早会对不上。
所有 {{占位符}} 都从 data/mock.json 的 meta 读, 改阈值只需要改生成器再重跑,
文档会跟着变 —— "看板与文档引用同一份配置"这句话才算数。

用法: python3 scripts/build_docs.py
"""

import json
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
META = json.loads((ROOT / "data" / "mock.json").read_text(encoding="utf-8"))["meta"]
TH, ACC = META["thresholds"], META["hll_accuracy"]


def pct(v, digits=0):
    return f"{v * 100:.{digits}f}%"


VALUES = {
    "success_rate_min": pct(TH["success_rate_min"]),
    "avg_gen_seconds_max": f"{TH['avg_gen_seconds_max']}",
    "reject_rate_max": pct(TH["reject_rate_max"]),
    "online_rate_min": pct(TH["online_rate_min"]),
    "scan_rate_min": pct(TH["scan_rate_min"]),
    "open_hour": str(META["open_hour"]),
    "close_hour": str(META["close_hour"]),
    "expected_minutes": str(META["expected_minutes_per_day"]),
    "hist_buckets": str(META["hist_buckets"]),
    "hll_registers": str(ACC["registers"]),
    "hll_median_err": f"{ACC['median_abs_err_pct']}%",
    "hll_p95_err": f"{ACC['p95_abs_err_pct']}%",
    "hll_max_err": f"{ACC['max_abs_err_pct']}%",
    "hll_slices": str(ACC["slices_tested"]),
    "date_from": META["date_range"][0],
    "date_to": META["date_range"][1],
}


def main():
    built = 0
    for tpl in sorted((ROOT / "docs").glob("*.template.md")):
        text = tpl.read_text(encoding="utf-8")
        missing = [k for k in re.findall(r"\{\{(\w+)\}\}", text) if k not in VALUES]
        if missing:
            print(f"错误: {tpl.name} 引用了未定义的占位符 {sorted(set(missing))}")
            return 1
        used = sorted(set(re.findall(r"\{\{(\w+)\}\}", text)))
        for k, v in VALUES.items():
            text = text.replace("{{" + k + "}}", v)
        out = tpl.with_name(tpl.name.replace(".template.md", ".md"))
        out.write_text(text, encoding="utf-8")
        print(f"{tpl.name}  →  {out.name}   注入 {len(used)} 个占位符: {', '.join(used)}")
        built += 1
    if not built:
        print("没有找到 *.template.md")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

# AIGC 互动大屏运营看板 · 指标说明

数据粒度 **天 × 点位**。每个数字都出自唯一的聚合函数 `computeMetrics()`：先按筛选取明细再聚合，**不是对已算好的比率再求平均**。本文的阈值全部从 `mock.json` 的 `meta` 注入（`scripts/build_docs.py`），改配置文档会跟着变。

## 核心指标

| 指标 | 定义 | 计算口径（字段见 `mock.json`） | 阈值 | 越界后该做什么 |
|---|---|---|---|---|
| **参与人数** | 两个口径分开看 | **去重人数**＝合并各格 participants_hll 草图后估算，跨天跨点位不重复计人；**人次**＝Σ participants（触屏会话数）。人均次数＝人次÷去重人数 | — | 人次跌超 20% 先查设备离线；人均次数升高说明复玩变好，可加大素材更新频率 |
| **生成成功率** | 成功出图的请求占比 | Σ gen_success ÷ Σ gen_requests（**次数**口径，一人多次算多次） | < {{success_rate_min}} | 拉失败请求 trace 定位模型版本，必要时回滚并暂停该点位新玩法 |
| **扫码 / 分享转化率** | 参与者中扫码带走、分享出去的占比 | Σ qr_scans ÷ Σ participants、Σ shares ÷ Σ participants。分母是**人次**，不是去重人数 | 扫码 < {{scan_rate_min}} | 扫码低：查取码步骤是否过长、二维码是否与出图同屏。扫码高而分享低：优化分享文案与落地页 |
| **平均 / P95 生成时长** | 一次成功生成从点击到出图的秒数 | 均值＝Σ gen_seconds_total ÷ Σ gen_success，**失败请求不计入**（多为超时，会拉高均值）。P95 由 gen_seconds_hist 合并后线性插值，是估算值 | 均值 > {{avg_gen_seconds_max}}s | 查该点位到推理集群的链路与队列深度；高峰期降分辨率或扩容并发。P95 远高于均值＝长尾卡顿，排查单点位而非整体扩容 |
| **审核通过率 / 拒绝率** | 机审 + 人审的处置结果占比 | Σ review_pass ÷ Σ review_total、Σ review_reject ÷ Σ review_total；送审量＝成功生成数 | 拒绝率 > {{reject_rate_max}} | 抽查被拒素材：是提示词诱导违规还是审核规则过严；相应改玩法文案或送审阈值 |
| **设备在线率** | 营业时段内实际在线时长占比 | Σ online_minutes ÷ Σ expected_minutes，每台每天应在线 {{expected_minutes}} 分钟（{{open_hour}}:00–{{close_hour}}:00） | < {{online_rate_min}} | 联系点位值守现场复位并查取电/网络；超 2 小时未恢复升级运维工单 |

**去重人数与 P95 都是估算值。** 人数和分位数都不可加，所以每格存可合并的结构而非标量：人数用 HyperLogLog（{{hll_registers}} 寄存器），时长用 {{hist_buckets}} 档直方图。HLL 在 {{hll_slices}} 个切片实测误差中位 {{hll_median_err}}、最大 {{hll_max_err}}，并用已知上下界夹逼；P95 因桶内插值假设均匀而真实时长右偏，恒略高于真值（实测 +0.3s）。两者看趋势、比点位可用，不作对外结算口径。

**环比**：与选定范围**等长且紧邻其前**的一段相比。比率用百分点（pp），计数用百分比，时长用秒差；前置数据不足显示“环比 —”。

## 异常查看的两条线

- **规则预警**：按上表阈值实时扫描当前切片。两种时间语义分开标注 ——「截至 {{date_to}} 未恢复」（拿数据集最后一天判，现在就得派单）与「范围末日越界」（只是区间末尾越界，可能早已恢复）。每条附越界天数，区分偶发抖动与长期问题。
- **异常事件**：设备与链路侧的离散故障。持续型异常按**连续区间**开单（带起止日期），表格按**区间相交**过滤 —— {{date_from}} 起持续两个月的问题，筛到任意中间片段都查得到。可按级别 / 类型 / 状态筛选，点击展开处理建议。

## 三、本人完成的主要修改

AI 的初稿能跑，但**跑得通不等于对**。下面是我自己发现并改掉的问题，按性质分三类。

### 1. 指标口径错误（影响数字正确性）

- **平均生成时长把失败请求也算了进去**。失败多半是超时，把它们计入均值会把数字系统性拉高。我改成只统计成功生成，并在指标说明里写明这条口径。
- **P95 原本按天存一个值，聚合时再求平均**——分位数不能这么算。我改成每条记录存一个 13 档的生成时长**直方图**，跨切片时先合并直方图再线性插值。注意插值出来的是**估算值**不是精确分位数——这一点我一开始写成了"都是准的"，后面被评审指出来才改。
- **转化率的分子分母混了口径**。生成成功率是"次数"口径（一人可以生成多次），扫码/分享转化率是"人数"口径。我在数据里拆成两组字段，并在文档里逐条标注。

### 2. 运营视角的缺陷（数字对，但没用）

- **告警按天刷屏**。初版对每个越界的日子都生成一条事件，一个持续 60 天的问题产生了 60 条记录，异常列表完全没法看。我改成按**连续区间去重**：一次持续异常只开一张单，详情里写持续天数——这也是真实告警系统的做法。
- **30 天平均把正在发生的事故抹平了**。有个点位最近 5 天成功率跌到 84%，但 30 天聚合后是 94%，看板一条预警都不报。我给规则引擎加了第二层判定：区间合计和**范围内最后一天**各判一次，只要最后一天越界就标记"仍在发生"并排到最前。这个改动让实时离线的设备和持续衰退的设备都浮了上来。
- **预警的处理建议给错了**。"扫码转化率偏低"复用了"素材下载失败"的处置文案，让人去查 CDN——但转化率低是现场引导问题，不是设备故障。我给这条规则单独写了建议动作。

### 3. 图表渲染缺陷（AI 自己看不见的）

这几个问题只有真的把页面打开看一眼才会发现，AI 写完代码是不会自己去看的：

- **百分比坐标轴显示成了 0.97、0.96**。原因是配置合并顺序写反了，公共轴配置覆盖掉了自定义的格式化函数。
- **阈值虚线根本不在可视范围内**。坐标轴自动缩放到数据范围（91%–97%），而阈值是 90%，线被画到了画布外面——卡片副标题还写着"虚线以下为低于阈值的日子"。我改成让轴的上下端点强制包含阈值。
- **坐标轴端点印出了 `12.471364357323855s`**。加留白后没取整。改成向外取整到干净的刻度。
- **两条趋势线几乎完全重合**，末端标签还互相压字。参与人数和生成请求次数本来就高度相关，同时画出来信息量接近零。我换成"参与人数 vs 扫码带走人数"——两条线量级拉得开，而且它们之间的距离直接就是流失掉的人，这才是运营想看的。
- **末端标签被卡片右边界切掉**。去掉了逐条末端标签，改成只标一个峰值，身份识别交给图例。

### 4. 其他工程决策

- **`file://` 下 `fetch()` 会被 CORS 拦**，评审方双击打开会白屏。数据生成脚本同时产出一份 `mock.js`（挂到 `window.MOCK_DATA`）用 `<script>` 引入，`mock.json` 作为交付物本身保留。ECharts 也放本地 `vendor/`，全程不依赖外网。
- **配色不靠眼睛判断**。分类色和漏斗的序数色都跑过对比度与色盲可分辨性校验；状态色（通过/拒绝/在线/离线）一律配文字标签，不让颜色单独承担含义；每张图都有表格视图，数值不依赖悬浮提示。
- **阈值集中在数据的 `meta.thresholds` 里**，看板和指标说明文档引用同一份，避免两边写着不同的数。

## 三点五、第三轮：让另一个模型做对抗性评审

自查只能发现"我想得到要查"的问题。所以做完之后，我把这份作业打包交给**另一个大模型**做盲评。

**做法**：把题目原文、全部源码逐字内联、以及用无头浏览器 dump 出来的真实渲染状态（三种筛选下的每个 KPI 数值、七张图的系列与阈值配置、设备表、异常表）做成一份自包含的材料，让它以资深面试官的身份评，并要求每条意见标注严重度、区分"事实错误"与"主观偏好"、凡是无法运行页面验证的结论必须声明是推测。开发过程的上下文单独放一个文件，让它**先盲评、后对照**，避免被我的叙述带跑。

**它提了六条事实性问题。我逐条复现，全部属实，没有一条误报**：

| 它指出的 | 我复现的结果 | 改法 |
|---|---|---|
| 文档称参与人数为"去重人数"，但数据只有天×点位汇总，跨天求和必然重复计人 | 属实，且是最致命的一条。30 天的 88,576 只能叫人次 | 给每条记录加了 **HyperLogLog 草图**：人数和分位数一样不可加，所以存可合并的结构而不是标量。现在去重人数在任意切片下都算得出来，顺带得到人均参与次数 |
| 持续型异常只在开始日建事件，事件表按开始日过滤 | 属实。一条 7/18 起持续 60 天的异常，在默认 30 天视图里预警在报、事件表却查不到 | 事件带起止日期，改为**按区间相交**过滤 |
| P95 是估算却被写成"都是准的"；"14 档"实为 13 档 | 属实。默认切片估算 15.7s vs 真实 15.39s；480 个单日切片最大偏差 +1.27s，且偏差**恒为正** | UI 与文档一律标注"估算"，并写明偏差恒为正的原因（桶内线性插值假设均匀，真实时长右偏） |
| "仍在发生"只代表所选范围末日越界；设备状态读数据集末日快照，却与所选日的当日指标并列 | 属实。筛到 9/5 时设备表显示某台"离线"（它 9/15 才断网），同行"当日参与"却是 9/5 的数 | 拆成「截至最新数据未恢复」与「范围末日越界」两个标记；设备表最新日用心跳、历史日按当天数据推导且心跳列置空 |
| 漏斗从"参与"直接跳到"生成成功"，已有的 gen_users 没用上 | 属实。合并的那一步里 24.4% 是压根没发起生成（引导问题）、6.3% 是发起了没成功（技术问题） | 漏斗补成五段，两类流失分开 |
| 自称"看板与文档引用同一份阈值"，实际文档里是手写字符串 | 属实，是过度声明 | 改成 `指标说明.template.md` + `build_docs.py`，阈值从数据的 meta 注入，这句话才成立 |

**修复过程中自己又发现两个问题**，都是因为写了校验才暴露的：

- HLL 用裸 FNV-1a 哈希时，对 `u0000001` 这类连续 ID 混合不足，**n=300 时估算偏低 49%**，寄存器下标卡方 2473（理想 1024）。加 murmur3 的 fmix32 收尾后最大误差降到 3.9%、卡方 1093。这种错不会报异常，只会安静地给出错的数字。
- 单日切片上 HLL 估出的去重人数**大于人次**，人均算成 0.98 次。用两个已知的硬约束把估算夹住（上界＝人次，下界＝单格去重最大值），并在只有一条记录时直接取精确值。

**校验脚本也按它的建议进了仓库**（它的原话是这比再加一张图更有说服力）：`scripts/verify.sh` 一条命令跑完数据可复现性、指标独立重算、渲染端对账共 60 项断言，其中包含它指出的每一个复现用例。

## 四、AI 帮了什么，没帮上什么

**帮上的**：把结构性的活儿压缩得很厉害——数据生成脚本、页面骨架、图表配置、文档初稿，这些如果手写要花掉大半天。让它顺手写一个 Playwright 自查脚本也很划算。

**没帮上的**：所有需要业务判断的地方它都会给一个"看起来合理"的答案，但未必对——平均时长算不算失败请求、告警要不要抑制、30 天平均会不会掩盖问题，这些只有真的想过运营每天怎么用这块看板才知道。渲染层面的问题它更是完全看不见，必须自己打开页面逐张图看。

**更值得说的是它的盲区**：写代码的那个模型自查了一轮，没发现"去重人数"这个口径根本立不住——因为它只会检查自己写的逻辑对不对，不会回头质疑口径本身的前提。这个问题是换一个模型、换一个视角（面试官而不是实现者）才暴露出来的。

我的用法是三段：**让 AI 写第一版 → 换一个模型以对抗视角挑错 → 我逐条复现后再决定改不改**。中间那一步不能省，最后一步也不能省：评审意见同样需要验证，只是这次六条全中而已。真正不可替代的是最后的判断——决定哪些该修、修到什么程度、以及像"去重人数"这种问题是改称谓还是改数据模型。我选了后者，因为面试官一定会追问"那怎么才能拿到真去重"，而这个答案应该写在代码里，不是靠嘴说。

| 1 | **"去重人数"口径立不住**。数据只有天×点位汇总，无身份字段，跨天跨点位求和必然重复计人 | 每条记录增加 **HyperLogLog 草图**（1024 寄存器）。人数与分位数一样不可加，故存可合并的结构而非标量。KPI 改为主显去重人数、副标题给人次与人均次数 | `verify.sh` 第 1/5 节：去重人数与独立实现一致；不变量检查确保去重 ≤ 人次、人均 ≥ 1 |
| 2 | **持续异常查不到**。事件只记开始日并按开始日过滤，2026-07-18 起持续两个月的异常在近 30 天视图里预警在报、事件表却空着 | 事件增加起止日期与 ongoing 标记，表格改为**区间相交**过滤，并加"持续期间"列 | `verify.sh` 第 3 节：7/14/30/60 天四种视图下都断言能查到该事件 |

| 4 | **时间语义混用**。"仍在发生"实为所选范围末日越界；设备状态永远读数据集末日快照，却与所选日的当日指标并排 | 拆成「截至 2026-09-15 未恢复」与「范围末日越界」两个标记；设备表最新日用心跳快照（能反映 20:32 才断网这种情况），历史日按当天数据推导且心跳列置空，列头标注日期 | `verify.sh` 第 4 节：7 项断言覆盖两种日期下的状态、列头与心跳列 |
| 5 | **漏斗合并了两类流失**。从"参与"直接跳到"生成成功"，已有的 gen_users 未使用 | 漏斗补成五段。现在能分开看：没发起生成（引导问题）与发起后失败（技术问题） | `verify.sh` 第 2 节：五段与独立实现一致且单调递减 |
| 6 | **"单一来源"是过度声明**。文档里的阈值是手写字符串，改数据不会同步 | 改为 `指标说明.template.md` + `scripts/build_docs.py`，阈值等 16 个数字从 `mock.json` 的 meta 注入 | 改生成器里的阈值 → 重跑 → 文档数字随之变化 |

## 修复过程中自己又发现的两个问题

都是因为写了校验才暴露出来的，不写就会安静地输出错误数字。

- **哈希混合不足**。HLL 用裸 FNV-1a 时，对 `u0000001` 这类连续 ID，n=300 的估算**偏低 49%**，寄存器下标卡方 2473（理想 1024）。加 murmur3 的 fmix32 收尾后，最大误差 3.9%、卡方 1093。
- **估算越界**。单日切片上估出的去重人数大于人次，人均显示成 0.98 次。改为用已知硬约束夹逼（上界＝人次，下界＝单格去重最大值），且只有一条记录时直接取精确值。

## 未解决 / 有意不做

- 转化率的分母仍是**人次**而非去重人数：给扫码、分享各配一份草图会让数据体积翻数倍，收益不匹配。文档已写明分母口径。
- 去重人数与 P95 都是**估算值**，误差已实测并写进文档（HLL 在 200 个切片上中位 1.74%、最大 6.66%）。不作对外结算口径。
- 仍无后端、权限模型与派单闭环；无单元测试（正确性依赖 `verify.sh` 的 60 项断言）；未做完整无障碍审计。
- 阈值（90% / 12s / 15% / 95% / 30%）仍是拍的，没有行业基准或历史基线支撑。

## 几个实现上的取舍

- **P95 存的是直方图不是数值**。分位数无法像均值那样加权聚合，每条记录存 14 档生成时长分布，跨切片时先合并直方图再线性插值，保证任意筛选组合下 P95 都算得准。
- **平均生成时长只统计成功生成**。失败多为超时，计入会系统性拉高均值。
- **告警按连续区间去重**。一次持续异常只开一张单并记录起止日期与持续天数，而不是每天刷一条。
- **不使用双轴图**。两个量纲不同的指标一律拆成两张图，避免用坐标轴的对齐方式编造出数据里没有的相关性。
- **颜色不单独承担含义**。状态一律配文字标签，每张图都提供表格视图。

- **文档里的阈值不手写**。`docs/*.template.md` 用占位符，由 `scripts/build_docs.py` 从 `mock.json` 的 `meta` 注入，改配置文档会跟着变。

````javascript
#!/usr/bin/env node
/**
 * docs/*.md  →  docs/*.pdf
 *
 * 用系统里已有的 Chromium 打印 A4。不引入 marked 之类的依赖, 这里自带一个
 * 够用的 Markdown 渲染(标题/表格/列表/引用/粗体/行内代码/链接), 文档本身
 * 也只用到这些语法。
 *
 *   node scripts/build_pdf.mjs
 */
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { chromium } from "playwright";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const DOCS = path.join(ROOT, "docs");

const esc = s => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
/* 行内: `代码` → **粗体** → [文字](链接) */
function inline(s) {
  return esc(s)
    .replace(/`([^`]+)`/g, (_, c) => "<code>" + c + "</code>")
    .replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>")
    .replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2">$1</a>');
}
const cells = row => row.replace(/^\||\|$/g, "").split("|").map(c => c.trim());

function md2html(src) {
  const lines = src.split("\n");
  const out = [];
  let inList = false, inTable = false;
  const closeList = () => { if (inList) { out.push("</ul>"); inList = false; } };
  const closeTable = () => { if (inTable) { out.push("</tbody></table>"); inTable = false; } };

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];

    // 表格: | a | b |  紧跟一行 |---|---|
    if (!inTable && /^\|.*\|$/.test(line.trim()) && /^\|[\s:|-]+\|$/.test((lines[i + 1] || "").trim())) {
      closeList();
      out.push("<table><thead><tr>"
        + cells(line.trim()).map(c => "<th>" + inline(c) + "</th>").join("")
        + "</tr></thead><tbody>");
      inTable = true; i++; continue;
    }
    if (inTable) {
      if (/^\|.*\|$/.test(line.trim())) {
        out.push("<tr>" + cells(line.trim()).map(c => "<td>" + inline(c) + "</td>").join("") + "</tr>");
        continue;
      }
      closeTable();
    }

    const h = line.match(/^(#{1,4})\s+(.*)$/);
    if (h) { closeList(); out.push(`<h${h[1].length}>${inline(h[2])}</h${h[1].length}>`); continue; }

    const li = line.match(/^\s*(?:[-*]|\d+\.)\s+(.*)$/);
    if (li) {
      if (!inList) { out.push("<ul>"); inList = true; }
      out.push("<li>" + inline(li[1]) + "</li>"); continue;
    }
    closeList();

    if (/^>\s?/.test(line)) { out.push("<blockquote>" + inline(line.replace(/^>\s?/, "")) + "</blockquote>"); continue; }
    if (/^```/.test(line)) {
      const buf = []; i++;
      while (i < lines.length && !/^```/.test(lines[i])) buf.push(lines[i++]);
      out.push("<pre><code>" + esc(buf.join("\n")) + "</code></pre>"); continue;
    }
    if (!line.trim()) continue;
    out.push("<p>" + inline(line) + "</p>");
  }
  closeList(); closeTable();
  return out.join("\n");
}

/* 打印样式。字体用系统里装了的文泉驿正黑, 保证中文不出豆腐块。 */
const CSS = `
@page { size: A4; margin: 12mm 11mm; }
*{box-sizing:border-box}
body{font-family:"WenQuanYi Zen Hei","Noto Sans CJK SC","PingFang SC",system-ui,sans-serif;
  font-size:8.5pt; line-height:1.42; color:#111; margin:0;}
h1{font-size:14pt; margin:0 0 2.5mm; padding-bottom:1.5mm; border-bottom:1.5pt solid #2a78d6}
h2{font-size:10.2pt; margin:3mm 0 1.2mm; color:#184f95}
h3{font-size:10pt; margin:4mm 0 1.5mm}
h4{font-size:9.5pt; margin:3mm 0 1mm}
p{margin:0 0 2mm}
ul{margin:0 0 2mm; padding-left:4.5mm}
li{margin:0 0 1.2mm}
table{width:100%; border-collapse:collapse; margin:1.5mm 0 2.5mm; font-size:7.5pt;
  table-layout:fixed; word-break:break-word}
th,td{border:0.5pt solid #c9c9c4; padding:1mm 1.4mm; text-align:left; vertical-align:top}
td code{white-space:normal; word-break:normal}
th{background:#eef4fc; font-weight:600; color:#184f95}
code{font-family:ui-monospace,Menlo,Consolas,monospace; font-size:7.6pt;
  background:#f1f1ee; padding:0 1mm; border-radius:2px}
pre{background:#f7f7f5; border:0.5pt solid #e1e0d9; border-radius:3px; padding:2mm;
  margin:0 0 2.5mm; overflow:hidden}
pre code{background:none; padding:0; font-size:8pt; line-height:1.4}
blockquote{margin:2mm 0; padding:1.5mm 3mm; border-left:2pt solid #c3c2b7;
  color:#52514e; background:#faf9f6}
strong{font-weight:600}
a{color:#1c5cab; text-decoration:none}
h2,h3,h4{break-after:avoid}
table,pre,blockquote{break-inside:avoid}
`;

// 跳过 *.template.md —— 它们是 build_docs.py 的输入, 不是给人读的文档
const files = fs.readdirSync(DOCS)
  .filter(f => f.endsWith(".md") && !f.endsWith(".template.md"));
const browser = await chromium.launch();
const page = await browser.newPage();
for (const f of files) {
  const md = fs.readFileSync(path.join(DOCS, f), "utf8");
  const html = `<!doctype html><html lang="zh-CN"><head><meta charset="utf-8">
<title>${esc(f)}</title><style>${CSS}</style></head><body>${md2html(md)}</body></html>`;
  await page.setContent(html, { waitUntil: "load" });
  // A4 版心 (210-26)×(297-28)mm @96dpi, 用来判断内容占了几页
  await page.setViewportSize({ width: 711, height: 1031 });
  const h = await page.evaluate(() => document.body.scrollHeight);
  const fill = (h / 1031 * 100).toFixed(0);
  const pdf = path.join(DOCS, f.replace(/\.md$/, ".pdf"));
  await page.pdf({ path: pdf, format: "A4", printBackground: true });
  // DOC_PNG=/某个目录 时额外出一张按 A4 版心截的图, 用来肉眼检查字号与排版
  if (process.env.DOC_PNG) {
    await page.screenshot({ path: path.join(process.env.DOC_PNG,
      f.replace(/\.md$/, ".png")), fullPage: true });
  }
  const pages = (fs.readFileSync(pdf).toString("latin1").match(/\/Type\s*\/Page[^s]/g) || []).length;
  console.log(`${f}  →  ${path.basename(pdf)}  (${pages} 页, 内容占版心 ${fill}%, ${(fs.statSync(pdf).size / 1024).toFixed(0)} KB)`);
}
await browser.close();
````
